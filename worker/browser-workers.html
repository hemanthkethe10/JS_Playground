🧵 Web Workers - Browser Console Demo

Learn Web Workers by typing commands in the Developer Console

📌 How to Use This Page

  2. Go to the Console tab
  3. Copy and paste the code snippets below to experiment!
  4. Watch the output in the console

1️⃣ Create a Simple Worker (Inline using Blob)

// Create a worker from inline code (no separate file needed!)
const workerCode = `
  self.onmessage = function(e) {
    console.log('Worker received:', e.data);
    const result = e.data.toUpperCase();
    self.postMessage(result);
  };
`;

const blob = new Blob([workerCode], { type: 'application/javascript' });
const worker = new Worker(URL.createObjectURL(blob));

worker.onmessage = (e) => console.log('Main received:', e.data);
worker.postMessage('hello from main thread');

2️⃣ CPU-Intensive Task: Sequential vs Parallel

// Fibonacci function (CPU intensive)
function fib(n) {
  if (n <= 1) return n;
  return fib(n - 1) + fib(n - 2);
}

// SEQUENTIAL - Blocks the main thread
console.time('Sequential');
[40, 40, 40, 40].forEach((n, i) => {
  console.log(`Task ${i+1}: fib(${n}) =`, fib(n));
});
console.timeEnd('Sequential');

3️⃣ Run Tasks in Parallel with Web Workers

// Worker code that calculates fibonacci
const fibWorkerCode = `
  function fib(n) {
    if (n <= 1) return n;
    return fib(n - 1) + fib(n - 2);
  }
  
  self.onmessage = function(e) {
    const { taskId, value } = e.data;
    const start = performance.now();
    const result = fib(value);
    const time = Math.round(performance.now() - start);
    self.postMessage({ taskId, value, result, time });
  };
`;

// Create 4 workers
const blob = new Blob([fibWorkerCode], { type: 'application/javascript' });
const workerUrl = URL.createObjectURL(blob);

console.log('🚀 Starting PARALLEL execution...');
console.time('Parallel');

const tasks = [40, 40, 40, 40];
let completed = 0;

tasks.forEach((value, i) => {
  const w = new Worker(workerUrl);
  w.onmessage = (e) => {
    console.log(`✓ Task ${e.data.taskId}: fib(${e.data.value}) = ${e.data.result} (${e.data.time}ms)`);
    completed++;
    if (completed === tasks.length) {
      console.timeEnd('Parallel');
      console.log('🎉 All tasks complete!');
    }
    w.terminate();
  };
  w.postMessage({ taskId: i + 1, value });
});

4️⃣ Easy Copy-Paste: Complete Comparison

// 🔥 COPY THIS ENTIRE BLOCK AND PASTE IN CONSOLE

(async () => {
  const FIB_VALUE = 40;
  const TASK_COUNT = 4;
  
  // Fibonacci function
  function fib(n) {
    if (n <= 1) return n;
    return fib(n - 1) + fib(n - 2);
  }
  
  console.log('════════════════════════════════════════');
  console.log('   SEQUENTIAL vs PARALLEL COMPARISON');
  console.log('════════════════════════════════════════');
  console.log(`Task: Calculate fib(${FIB_VALUE}) × ${TASK_COUNT} times\n`);
  
  // SEQUENTIAL
  console.log('📊 SEQUENTIAL (Main Thread)...');
  const seqStart = performance.now();
  for (let i = 0; i < TASK_COUNT; i++) {
    const start = performance.now();
    const result = fib(FIB_VALUE);
    console.log(`  Task ${i+1}: ${result} (${Math.round(performance.now() - start)}ms)`);
  }
  const seqTime = Math.round(performance.now() - seqStart);
  console.log(`  ⏱ Total: ${seqTime}ms\n`);
  
  // PARALLEL
  console.log('⚡ PARALLEL (Web Workers)...');
  const workerCode = `
    function fib(n) { if (n <= 1) return n; return fib(n-1) + fib(n-2); }
    self.onmessage = (e) => {
      const start = performance.now();
      const result = fib(e.data.value);
      self.postMessage({ id: e.data.id, result, time: Math.round(performance.now() - start) });
    };
  `;
  
  const blob = new Blob([workerCode], { type: 'application/javascript' });
  const url = URL.createObjectURL(blob);
  
  const parStart = performance.now();
  const promises = Array.from({ length: TASK_COUNT }, (_, i) => {
    return new Promise((resolve) => {
      const w = new Worker(url);
      w.onmessage = (e) => { 
        console.log(`  Task ${e.data.id}: ${e.data.result} (${e.data.time}ms)`);
        w.terminate(); 
        resolve(e.data); 
      };
      w.postMessage({ id: i + 1, value: FIB_VALUE });
    });
  });
  
  await Promise.all(promises);
  const parTime = Math.round(performance.now() - parStart);
  console.log(`  ⏱ Total: ${parTime}ms\n`);
  
  // COMPARISON
  const speedup = (seqTime / parTime).toFixed(2);
  console.log('════════════════════════════════════════');
  console.log(`  Sequential: ${seqTime}ms`);
  console.log(`  Parallel:   ${parTime}ms`);
  console.log(`  Speedup:    ${speedup}x faster!`);
  console.log('════════════════════════════════════════');
  
  URL.revokeObjectURL(url);
})();

📚 Quick Reference

// Create worker from inline code
const blob = new Blob(['self.onmessage = (e) => { ... }'], { type: 'application/javascript' });
const worker = new Worker(URL.createObjectURL(blob));

// Send message to worker
worker.postMessage({ data: 'hello' });

// Receive message from worker
worker.onmessage = (e) => console.log(e.data);

// Handle errors
worker.onerror = (e) => console.error(e.message);

// Terminate worker when done
worker.terminate();
